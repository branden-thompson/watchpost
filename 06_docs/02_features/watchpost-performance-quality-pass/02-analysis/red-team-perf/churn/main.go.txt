package main

import (
	"fmt"
	"os"
	"os/exec"
	"runtime"
	"strconv"
	"strings"
	"time"
)

var sink [][]byte

func rss() string {
	out, _ := exec.Command("ps", "-o", "rss=", "-p", strconv.Itoa(os.Getpid())).Output()
	kb, _ := strconv.Atoi(strings.TrimSpace(string(out)))
	return fmt.Sprintf("%d MB", kb/1024)
}
func footprint() string {
	out, _ := exec.Command("vmmap", "--summary", strconv.Itoa(os.Getpid())).Output()
	for _, l := range strings.Split(string(out), "\n") {
		if strings.HasPrefix(l, "Physical footprint:") { return strings.TrimSpace(strings.TrimPrefix(l, "Physical footprint:")) }
	}
	return "?"
}

func main() {
	rateMBmin, _ := strconv.ParseFloat(os.Args[1], 64)
	secs, _ := strconv.Atoi(os.Args[2])
	// retain ~42 MB live like the instrumented process
	live := make([][]byte, 0, 42)
	for i := 0; i < 42; i++ { b := make([]byte, 1<<20); for j := range b { b[j] = 1 }; live = append(live, b) }
	runtime.GC()
	var ms runtime.MemStats
	runtime.ReadMemStats(&ms)
	fmt.Printf("rate=%.0f MB/min start: HeapInuse=%d MB HeapSys=%d MB rss=%s footprint=%s\n", rateMBmin, ms.HeapInuse>>20, ms.HeapSys>>20, rss(), footprint())
	// churn: 437 KB per frame, per-frame allocs of small objects like a render (10k allocs of ~44 B) — approximate with 1000 x 437 B
	perFrame := 437 * 1024.0
	fps := rateMBmin * 1024 * 1024 / 60 / perFrame
	interval := time.Duration(float64(time.Second) / fps)
	end := time.Now().Add(time.Duration(secs) * time.Second)
	for time.Now().Before(end) {
		frame := make([][]byte, 0, 1000)
		for i := 0; i < 1000; i++ { frame = append(frame, make([]byte, 437)) }
		sink = frame
		time.Sleep(interval)
	}
	runtime.ReadMemStats(&ms)
	fmt.Printf("rate=%.0f MB/min end:   HeapInuse=%d MB HeapSys=%d MB NumGC=%d rss=%s footprint=%s\n", rateMBmin, ms.HeapInuse>>20, ms.HeapSys>>20, ms.NumGC, rss(), footprint())
	_ = live[0][0]
}
