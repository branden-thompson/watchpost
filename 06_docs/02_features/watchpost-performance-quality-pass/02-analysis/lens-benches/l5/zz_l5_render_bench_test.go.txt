func l5Rows(n int) []LocationRow {
	rows := make([]LocationRow, 0, n)
	for i := range n {
		r := LocationRow{Index: i + 1, Name: fmt.Sprintf("Benchmark City %02d", i), Zip: "92057", Station: "KCRQ", StationKM: fp(12.5),
			Conditions: "partly_cloudy", Now: fp(22.8), Hi: fp(30), Lo: fp(17), Trend: "up",
			TomorrowConditions: "clear", TomorrowHi: fp(31), TomorrowLo: fp(18), Selected: i == 2, HasAlert: i%3 == 0, WarnAlert: i%6 == 0, Fire: i % 4}
		for d := range 5 {
			r.Extended = append(r.Extended, DayCell{Date: fmt.Sprintf("08/%02d", 26+d), Hi: fp(30 + float64(d)), Lo: fp(18)})
		}
		rows = append(rows, r)
	}
	return rows
}

func fp(v float64) *float64 { return &v }

func BenchmarkL5LocationTable_50rows_w129(b *testing.B) {
	rendering.SetColorEnabledForTest(true)
	defer rendering.ResetColorEnabledForTest()

	}
}

// Width agreement table: what the app measures vs what the renderer (ultraviolet/x/ansi) lays out.
func TestL5WidthAgreement(t *testing.T) {
	samples := []string{"◆", "▶", "⚠", "✔", "✘", "º", "∞", "↗", "↘", "♪", "■", "…", "━", "█", "▲", "▼", "│", "›", "─", "┌", "⠋",
		"⚠️", "é", "東京", "a​b", "👍", "👨‍👩‍👧", "🕐", "ºF", "22ºC", "\x1b[38;5;208m 72ºF\x1b[0m"}
	t.Logf("%-14s runewidth uniseg displaywidth x/ansi.StringWidth kit.DisplayWidth app.displayWidth", "sample")
	for _, s := range samples {
		t.Logf("%-14q %9d %6d %12d %18d %16d %16d", s, runewidth.StringWidth(s), uniseg.StringWidth(s), displaywidth.String(s), ansi.StringWidth(s), rendering.DisplayWidth(s), displayWidth(s))
	}
	t.Logf("runewidth EastAsian=%v", runewidth.DefaultCondition.EastAsianWidth)
}

// Composite token behaviour through the kit's classifier (UAT 108 rule).
func TestL5ColorSequenceComposites(t *testing.T) {
	for _, c := range []string{"208", "1;208", "1;38;5;220", "38;5;250", "48;2;66;94;122", "1;97;48;2;86;86;86", "97", "1;97", "38;2;190;84;84"} {
		t.Logf("%-24q -> SGR=%q", c, rendering.SGR(c))
	}
}
