func l5Loc(i int, days int, alert bool) snapshot.Location {
	loc := snapshot.Location{
		Label: fmt.Sprintf("Benchmark City %02d", i+1), Zip: fmt.Sprintf("9%04d", i),
		Lat: 33.0 + float64(i)/10, Lon: -117.0 - float64(i)/10,
		Harmonized: snapshot.Conditions{Temp: f64(22.8 + float64(i%7)), Condition: "partly_cloudy",
			Source: snapshot.SourceInfo{Provider: "nws", ModelOrStation: "KCRQ", DistanceKm: f64(12.5)}},
	}
	for d := range days {
		loc.Daily = append(loc.Daily, snapshot.Daily{Date: fmt.Sprintf("2026-08-%02d", 24+d), TempMax: f64(23.9 + float64(d)), TempMin: f64(17.2), Condition: "clear"})
	}
	if alert {
		loc.Alerts = []snapshot.Alert{{Event: "Extreme Heat Watch", Severity: "severe", Headline: "until Friday"}}
	}
	return loc
}

func l5Dash(b *testing.B, w, h, prio, recent int) tea.Model {
	b.Helper()
	m, err := NewDashboard(Config{Version: "0.9.4-bench"})
	if err != nil {
		b.Fatal(err)
	}
	obs := time.Date(2026, 8, 24, 1, 0, 0, 0, time.UTC)
	sn := &snapshot.Snapshot{SchemaVersion: snapshot.SchemaVersion, GeneratedAt: obs,
		Providers: []snapshot.ProviderStatus{{ID: "nws", Status: snapshot.ProviderOK, FetchedAt: obs}, {ID: "firms", Status: snapshot.ProviderOK, FetchedAt: obs}}}
	for i := range prio {
		sn.Locations = append(sn.Locations, l5Loc(i, 7, i%3 == 0))
	}
	rs := &snapshot.Snapshot{SchemaVersion: snapshot.SchemaVersion, GeneratedAt: obs}
	for i := range recent {
		rs.Locations = append(rs.Locations, l5Loc(100+i, 7, i%5 == 0))
	}
	var model tea.Model = m
	model, _ = model.Update(tea.WindowSizeMsg{Width: w, Height: h})
	model, _ = model.Update(SnapshotMsg{Snap: sn})
	model, _ = model.Update(RecentSnapshotMsg{Snap: rs})
	return model
}

func benchView(b *testing.B, color bool, w, h, prio, recent int, help bool) {
	rendering.SetColorEnabledForTest(color)
	defer rendering.ResetColorEnabledForTest()
	m := l5Dash(b, w, h, prio, recent)
	if help {
		m, _ = m.Update(tea.KeyPressMsg{Code: '?', Text: "?"})
	}
	v := m.View().Content
	b.ReportMetric(float64(len(v)), "bytes/frame")
	b.ResetTimer()
	b.ReportAllocs()
	for range b.N {
		_ = m.View().Content
	}
}

func BenchmarkL5View_Color_133x44_10p50r(b *testing.B) { benchView(b, true, 133, 44, 10, 50, false) }
func BenchmarkL5View_NoColor_133x44_10p50r(b *testing.B) {
	benchView(b, false, 133, 44, 10, 50, false)
}
func BenchmarkL5View_Color_133x44_10p50r_Help(b *testing.B) {
	benchView(b, true, 133, 44, 10, 50, true)
}
func BenchmarkL5View_Color_200x60_10p50r(b *testing.B) { benchView(b, true, 200, 60, 10, 50, false) }
func BenchmarkL5View_Color_133x44_1p1r(b *testing.B)   { benchView(b, true, 133, 44, 1, 1, false) }

// Same-frame identity: does View() return byte-identical output when nothing changed?
func TestL5ViewStable(t *testing.T) {
	rendering.SetColorEnabledForTest(true)
	defer rendering.ResetColorEnabledForTest()
	m, err := NewDashboard(Config{Version: "x"})
	if err != nil {
		t.Fatal(err)
	}
	var model tea.Model = m
	model, _ = model.Update(tea.WindowSizeMsg{Width: 133, Height: 44})
	model, _ = model.Update(SnapshotMsg{Snap: snap()})
	a := model.View().Content
	b := model.View().Content
	t.Logf("frame bytes=%d lines=%d identical=%v", len(a), countNL(a), a == b)
	model, _ = model.Update(tickMsg{})
	c := model.View().Content
	t.Logf("after tick (frame++): identical=%v", a == c)
}

func countNL(s string) int {
	n := 0
	for _, r := range s {
		if r == '\n' {
			n++
		}
	}
	return n
}
